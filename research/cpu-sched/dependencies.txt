# k8labs CPU Research — Required Dependencies
# ==========================================
# Run: xargs -a dependencies.txt sudo apt-get install -y
# or manually install each package.

# --- Core tools ---

tofu | terraform                 # OpenTofu or Terraform (or via mise)

ssh                              # OpenSSH client
jq                               # JSON query tool
yq                               # YAML query tool (optional, for advanced parsing)

python3                          # Python 3.x
python3-matplotlib               # Plotting library
python3-numpy                    # Numerical computing

golang                           # Go compiler (optional, pre-built binary included)
